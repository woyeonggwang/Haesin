// 빌드: gcc -O3 -shared -fPIC voxel.c -o libvoxel.so -lm  (Unity 가 .c 를 네이티브 소스로 보므로 .c.txt 로 둠)
// 삼각형 표면을 촘촘한 점(간격 h)으로 찍어 복셀을 막음(1)으로 표시. grid 는 (z*NY + y)*NX + x.
#include <stdint.h>
#include <math.h>
static inline void mark(uint8_t* g, const float* o, float v, int NX, int NY, int NZ, float px, float py, float pz, uint8_t val) {
    int x = (int)floorf((px - o[0]) / v), y = (int)floorf((py - o[1]) / v), z = (int)floorf((pz - o[2]) / v);
    if (x < 0 || y < 0 || z < 0 || x >= NX || y >= NY || z >= NZ) return;
    g[((long long)z * NY + y) * NX + x] = val;
}
long long voxelize(const float* V, const int* F, int nt, const float* o, float v, int NX, int NY, int NZ, float h, uint8_t* g, uint8_t val) {
    long long pts = 0;
    for (int t = 0; t < nt; t++) {
        const float* a = V + 3 * F[3 * t]; const float* b = V + 3 * F[3 * t + 1]; const float* c = V + 3 * F[3 * t + 2];
        float e0 = sqrtf((b[0]-a[0])*(b[0]-a[0]) + (b[1]-a[1])*(b[1]-a[1]) + (b[2]-a[2])*(b[2]-a[2]));
        float e1 = sqrtf((c[0]-b[0])*(c[0]-b[0]) + (c[1]-b[1])*(c[1]-b[1]) + (c[2]-b[2])*(c[2]-b[2]));
        float e2 = sqrtf((a[0]-c[0])*(a[0]-c[0]) + (a[1]-c[1])*(a[1]-c[1]) + (a[2]-c[2])*(a[2]-c[2]));
        float me = e0 > e1 ? (e0 > e2 ? e0 : e2) : (e1 > e2 ? e1 : e2);
        int m = (int)ceilf(me / h); if (m < 1) m = 1;
        for (int i = 0; i <= m; i++) for (int j = 0; j <= m - i; j++) {
            float u = (float)i / m, w = (float)j / m, s = 1.0f - u - w;
            mark(g, o, v, NX, NY, NZ, s*a[0] + u*b[0] + w*c[0], s*a[1] + u*b[1] + w*c[1], s*a[2] + u*b[2] + w*c[2], val);
            pts++;
        }
    }
    return pts;
}
// 점 목록을 복셀로(알파 판처럼 미리 거른 점)
void mark_points(const float* P, long long n, const float* o, float v, int NX, int NY, int NZ, uint8_t* g, uint8_t val) {
    for (long long i = 0; i < n; i++) mark(g, o, v, NX, NY, NZ, P[3*i], P[3*i+1], P[3*i+2], val);
}
